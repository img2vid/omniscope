    48→export interface DiffHunk {
    49→  start: number;
    50→  /** exclusive end */
    51→  end: number;
    52→  /** true when this hunk represents the tail of the longer file (past the shorter file's EOF) */
    53→  tail: boolean;
    54→}

    56→/**
    57→ * Compute contiguous differing regions between two byte arrays.
    58→ * Runs closer than 16 bytes are merged into one hunk. The length difference
    59→ * (bytes past min(lenA, lenB)) becomes a final hunk flagged with tail=true.
    60→ */
    61→export function computeDiffHunks(a: Uint8Array, b: Uint8Array): DiffHunk[] {
    62→  const n = Math.min(a.length, b.length);
    63→  const hunks: DiffHunk[] = [];
    64→  let cur: DiffHunk | null = null;
    65→  let gap = 0;
    66→  for (let i = 0; i < n; i++) {
    67→    if (a[i] !== b[i]) {
    68→      if (cur && gap > 0 && gap < 16 && cur.end === i - gap) {
    69→        // extend current hunk across the small gap
    70→        cur.end = i + 1;
    71→      } else {
    72→        if (cur) hunks.push(cur);
    73→        cur = { start: i, end: i + 1, tail: false };
    74→      }
    75→      gap = 0;
    76→    } else if (cur) {
    77→      gap++;
    78→    }
    79→    if (hunks.length >= 5000) break;
    80→  }
    81→  if (cur) hunks.push(cur);
    82→  const maxLen = Math.max(a.length, b.length);
    83→  if (maxLen > n) hunks.push({ start: n, end: maxLen, tail: true });
    84→  return hunks;
    85→}

   104→async function sha256Chunks(file: File, cap: number, onProgress?: (done: number) => void): Promise<string> {
   105→  const limit = Math.min(file.size, cap);
   106→  const parts: ArrayBuffer[] = [];
   107→  const CH = 8 * 1024 * 1024;
   108→  for (let off = 0; off < limit; off += CH) {
   109→    const end = Math.min(off + CH, limit);
   110→    parts.push(await file.slice(off, end).arrayBuffer());
   111→    onProgress?.(Math.min(end, limit));
   112→  }
   113→  const total = parts.reduce((s, p) => s + p.byteLength, 0);
   114→  const merged = new Uint8Array(total);
   115→  let pos = 0;
   116→  for (const p of parts) { merged.set(new Uint8Array(p), pos); pos += p.byteLength; }
   117→  const dig = await crypto.subtle.digest("SHA-256", merged);
   118→  return Array.from(new Uint8Array(dig)).map((x) => x.toString(16).padStart(2, "0")).join("");
   119→}

   135→function buildTextSide(bytes: Uint8Array): TextSide {
   136→  const truncatedBytes = bytes.length > TEXT_CAP;
   137→  const slice = truncatedBytes ? bytes.subarray(0, TEXT_CAP) : bytes;
   138→  const dec = decodeTextual(slice);
   139→  const all = splitLines(dec.text);
   140→  const truncatedLines = all.length > LINE_CAP;
   141→  const lines = truncatedLines ? all.slice(0, LINE_CAP) : all;
   142→  return {
   143→    lines,
   144→    enc: dec.enc,
   145→    truncatedBytes,
   146→    truncatedLines,
   147→    noNewline: dec.text.length > 0 && !/[\r\n]$/.test(dec.text),
   148→  };
   149→}

   429→        i++;
   430→        continue;
   431→      }
   432→      const dels: number[] = [];
   433→      const ins: number[] = [];
   434→      while (i < ops.length && ops[i].type !== "equal") {
   435→        if (ops[i].type === "delete") dels.push(i);
   436→        else ins.push(i);
   437→        i++;
   438→      }
   439→      const pairs = Math.min(dels.length, ins.length);
   440→      for (let k = 0; k < pairs; k++) {
   441→        const d = ops[dels[k]] as Extract<LineOp, { type: "delete" }>;
   442→        const n = ops[ins[k]] as Extract<LineOp, { type: "insert" }>;
   443→        const w = wordDiff(textA.lines[d.aIdx], textB.lines[n.bIdx]);
   444→        rows.push({
   445→          kind: "mod",
   446→          aN: d.aIdx, aText: textA.lines[d.aIdx],
