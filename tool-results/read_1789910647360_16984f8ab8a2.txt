   103→function u64le(b: Uint8Array, o: number): number {
   104→  // 64-bit LE as a Number (precise up to 2^53 — enough for file offsets)
   105→  let v = 0;
   106→  for (let i = 7; i >= 0; i--) v = v * 256 + (b[o + i] ?? 0);
   107→  return v;
   108→}

   125→function dosDateTime(time: number, date: number): number | null {
   126→  if (!date && !time) return null;
   127→  const year = ((date >> 9) & 0x7f) + 1980;
   128→  const mon = ((date >> 5) & 0x0f) || 1;
   129→  const day = (date & 0x1f) || 1;
   130→  const h = (time >> 11) & 0x1f;
   131→  const m = (time >> 5) & 0x3f;
   132→  const s = (time & 0x1f) * 2;
   133→  const d = new Date(Date.UTC(year, mon - 1, day, h, m, s));
   134→  return isNaN(d.getTime()) ? null : d.getTime();
   135→}

   139→function parseZip(src: ByteSource): ArchiveModel {
   140→  const b = src.bytes;
   141→  const n = b.length;
   142→  const warnings: string[] = [];
   143→  const scanFrom = Math.max(0, n - 66000);
   144→  let eocd = -1;
   145→  for (let i = n - 22; i >= scanFrom; i--) {
   146→    if (b[i] === 0x50 && b[i + 1] === 0x4b && b[i + 2] === 0x05 && b[i + 3] === 0x06) { eocd = i; break; }
   147→  }
   148→  if (eocd < 0) {
   149→    throw new Error("ZIP: end of central directory not found — the file may be truncated or part of a multi-part set.");
   150→  }
   151→  let count = u16le(b, eocd + 10);
   152→  let cdSize = u32le(b, eocd + 12);
   153→  let cdOffset = u32le(b, eocd + 16);
   154→  const commentLen = u16le(b, eocd + 20);
   155→  const comment = commentLen ? latin1(b.subarray(eocd + 22, eocd + 22 + commentLen)).trim() : "";
   156→
   157→  // ZIP64 fallback
   158→  if (count === 0xffff || cdSize === 0xffffffff || cdOffset === 0xffffffff) {
   159→    for (let i = eocd - 20; i >= scanFrom; i--) {
   160→      if (b[i] === 0x50 && b[i + 1] === 0x4b && b[i + 2] === 0x06 && b[i + 3] === 0x07) {
   161→        const z64abs = u64le(b, i + 8);
   162→        const z = z64abs - src.base;
   163→        if (z >= 0 && z + 56 <= n && b[z] === 0x50 && b[z + 1] === 0x4b && b[z + 2] === 0x06 && b[z + 3] === 0x06) {
   164→          count = u64le(b, z + 32);
   165→          cdSize = u64le(b, z + 40);
   166→          cdOffset = u64le(b, z + 48);
   167→          warnings.push("ZIP64 layout (over 4 GB addressing) detected.");
   168→        }
   169→        break;
   170→      }
   171→    }
   172→  }
   173→
   174→  // locate the central directory inside the loaded window
   175→  let cd = -1;
   176→  for (const c of [cdOffset - src.base, cdOffset]) {
   177→    if (c >= 0 && c + 4 <= n && u32le(b, c) === 0x02014b50) { cd = c; break; }
   178→  }
   179→  if (cd < 0) {
   180→    const c = eocd - cdSize;
   181→    if (c >= 0 && c + 4 <= n && u32le(b, c) === 0x02014b50) {
   182→      cd = c;
   183→      warnings.push("Archive has prepended data (self-extracting / wrapped ZIP) — offsets adjusted.");
   184→    }
   185→  }
   186→  if (cd < 0) throw new Error("ZIP: central directory not present inside the loaded bytes.");
   187→  if (src.base > 0) {
   188→    warnings.push("Directory-only mode: the file body was not loaded (over the in-memory cap) — entry previews and downloads are unavailable.");
   189→  }
   190→
   191→  const entries: ArcEntry[] = [];
   192→  let off = cd;
   193→  let bad = 0;
   194→  const limit = Math.min(count, 100000);
   195→  for (let i = 0; i < limit; i++) {
   196→    if (off + 46 > n) break;
   197→    if (u32le(b, off) !== 0x02014b50) { bad++; break; }
   198→    const flags = u16le(b, off + 8);
   199→    const method = u16le(b, off + 10);
   200→    const time = u16le(b, off + 12);
   201→    const date = u16le(b, off + 14);
   202→    const crc = u32le(b, off + 16);
   203→    let packed = u32le(b, off + 20);
   204→    let size = u32le(b, off + 24);
   205→    const nameLen = u16le(b, off + 28);
   206→    const extraLen = u16le(b, off + 30);
   207→    const entryCommentLen = u16le(b, off + 32);
   208→    let local = u32le(b, off + 42);
   209→    const extAttr = u32le(b, off + 38);
   210→    if (off + 46 + nameLen > n) { bad++; break; }
   211→    const name = flags & 0x800
   212→      ? new TextDecoder().decode(b.subarray(off + 46, off + 46 + nameLen))
   213→      : decodeCP437(b.subarray(off + 46, off + 46 + nameLen));
   214→
   215→    // ZIP64 extra field
   216→    let eo = off + 46 + nameLen;
   217→    const eEnd = Math.min(eo + extraLen, n);
   218→    while (eo + 4 <= eEnd) {
   219→      const id = u16le(b, eo);
   220→      const sz = u16le(b, eo + 2);
   221→      if (id === 0x0001) {
   222→        let p = eo + 4;
   223→        if (size === 0xffffffff && p + 8 <= eEnd) { size = u64le(b, p); p += 8; }
   224→        if (packed === 0xffffffff && p + 8 <= eEnd) { packed = u64le(b, p); p += 8; }
   225→        if (local === 0xffffffff && p + 8 <= eEnd) { local = u64le(b, p); p += 8; }
   226→      }
   227→      eo += 4 + sz;
   228→    }
   229→
   230→    off += 46 + nameLen + extraLen + entryCommentLen;
   231→    if (!name) { bad++; continue; }
   232→    const isDir = name.endsWith("/") || (extAttr & 0x10) !== 0 || ((extAttr >>> 16) & 0xf000) === 0x4000;
   233→    const encrypted = (flags & 1) !== 0;
   234→    entries.push({
   235→      path: name,
   236→      isDir,
   237→      size,
   238→      packed,
   239→      method: ZIP_METHODS[method] ?? `method ${method}`,
   240→      crc: isDir ? null : crc,
   241→      encrypted,
   242→      mtime: dosDateTime(time, date),
   243→      read: isDir || encrypted ? undefined : () => readZipEntry(src, local, packed, method),
   244→    });
   245→  }
   246→  if (count > limit) warnings.push(`Entry list capped at ${limit} of ${count} records.`);
   247→  return { kind: "zip", entries, comment: comment || undefined, bad, warnings, tailOnly: src.base > 0 };
   248→}

   250→function readZipEntry(src: ByteSource, local: number, packed: number, method: number): Uint8Array<ArrayBuffer> {
   251→  const b = src.bytes;
   252→  const off = local - src.base;
   253→  if (off < 0 || off + 30 > b.length) {
   254→    throw new Error("Entry data lies outside the loaded byte window (file exceeds the in-memory cap).");
   255→  }
   256→  if (u32le(b, off) !== 0x04034b50) throw new Error("Local header mismatch — unsupported archive layout.");
   257→  const nameLen = u16le(b, off + 26);
   258→  const extraLen = u16le(b, off + 28);
   259→  const start = off + 30 + nameLen + extraLen;
   260→  if (start + packed > b.length) throw new Error("Compressed data incomplete inside the loaded window.");
   261→  const data = b.subarray(start, start + packed);
   262→  if (method === 0) return data;
   263→  if (method === 8) return inflateSync(data);
   264→  throw new Error(`Compression method “${ZIP_METHODS[method] ?? method}” is not decodable in-browser yet.`);
   265→}

   269→function parseOctal(field: Uint8Array): number {
   270→  if (field.length && (field[0] & 0x80) !== 0) {
   271→    // GNU base-256 extension
   272→    let v = field[0] & 0x7f;
   273→    for (let i = 1; i < field.length; i++) v = v * 256 + field[i];
   274→    return v;
   275→  }
   276→  let s = "";
   277→  for (let i = 0; i < field.length; i++) {
   278→    const c = field[i];
   279→    if (c === 0) break;
   280→    if (c !== 0x20) s += String.fromCharCode(c);
   281→  }
   282→  if (!/^[0-7]*$/.test(s)) return 0;
   283→  const v = parseInt(s, 8);
   284→  return isNaN(v) ? 0 : v;
   285→}

   287→function tarCStr(field: Uint8Array): string {
   288→  let s = "";
   289→  for (let i = 0; i < field.length; i++) {
   290→    const c = field[i];
   291→    if (c === 0) break;
   292→    s += String.fromCharCode(c);
   293→  }
   294→  return s;
   295→}

   297→function tarChecksumOk(h: Uint8Array): boolean {
   298→  let sum = 0;
   299→  for (let i = 0; i < 512; i++) sum += i >= 148 && i < 156 ? 32 : h[i];
   300→  const stored = parseOctal(h.subarray(148, 156));
   301→  return stored > 0 && sum === stored;
   302→}

   304→function parseTar(b: Uint8Array<ArrayBuffer>): { entries: ArcEntry[]; variant: string } {
   305→  const entries: ArcEntry[] = [];
   306→  let off = 0;
   307→  let longName: string | null = null;
   308→  let paxPath: string | null = null;
   309→  const magic = asciiAt(b, 257, 5);
   310→  const variant = magic === "ustar" ? (b[263] === 0x00 ? "ustar (POSIX)" : b[263] === 0x20 ? "ustar (GNU)" : "ustar") : "pre-POSIX (v7)";
   311→  const CAP = 25000;
   312→  while (off + 512 <= b.length && entries.length < CAP) {
   313→    const h = b.subarray(off, off + 512);
   314→    if (h[0] === 0 && h[1] === 0 && h[2] === 0 && h[511] === 0) break; // end-of-archive zero block
   315→    if (asciiAt(h, 257, 5) !== "ustar" && !tarChecksumOk(h)) break; // not a header
   316→    const type = h[156];
   317→    const size = parseOctal(h.subarray(124, 136));
   318→    const mtime = parseOctal(h.subarray(136, 148)) * 1000;
   319→    const dataStart = off + 512;
   320→    const name = longName ?? paxPath ?? (() => {
   321→      const base = tarCStr(h.subarray(0, 100));
   322→      const prefix = magic === "ustar" ? tarCStr(h.subarray(345, 500)) : "";
   323→      return prefix ? prefix + "/" + base : base;
   324→    })();
   325→    longName = null;
   326→    paxPath = null;
   327→
   328→    if (type === 0x4c) { // GNU long name — payload is the name of the NEXT entry
   329→      longName = tarCStr(b.subarray(dataStart, dataStart + Math.min(size, 4096)));
   330→      off = dataStart + Math.ceil(size / 512) * 512;
   331→      continue;
   332→    }
   333→    if (type === 0x78 || type === 0x67) { // pax extended / global header
   334→      const text = latin1(b.subarray(dataStart, dataStart + Math.min(size, 65536)));
   335→      const at = text.indexOf(" path=");
   336→      if (at >= 0) {
   337→        const rest = text.slice(at + 6);
   338→        const nl = rest.indexOf("\n");
   339→        paxPath = nl >= 0 ? rest.slice(0, nl) : rest;
   340→      }
   341→      off = dataStart + Math.ceil(size / 512) * 512;
   342→      continue;
   343→    }
   344→
   345→    const kind = type === 0 ? "0" : String.fromCharCode(type);
   346→    const isDir = kind === "5" || (name.endsWith("/") && kind === "0");
   347→    const special: Record<string, string> = { "1": "hard link", "2": "symlink", "3": "char device", "4": "block device", "6": "FIFO" };
   348→    entries.push({
   349→      path: name,
   350→      isDir,
   351→      size: isDir ? 0 : size,
   352→      packed: size,
   353→      method: isDir ? "directory" : kind === "0" || kind === "7" ? "stored" : special[kind] ?? `type ${kind}`,
   354→      crc: null,
   355→      encrypted: false,
   356→      mtime: mtime || null,
   357→      read: !isDir && (kind === "0" || kind === "7")
   358→        ? () => b.subarray(dataStart, Math.min(dataStart + size, b.length))
   359→        : undefined,
   360→    });
   361→    off = dataStart + Math.ceil(size / 512) * 512;
   362→  }
   363→  return { entries, variant };
   364→}

   372→function parseIso(b: Uint8Array<ArrayBuffer>): ArchiveModel {
   373→  let pvd = -1;
   374→  let joliet = false;
   375→  let off = 0x8000;
   376→  for (let i = 0; i < 16 && off + 2048 <= b.length; i++) {
   377→    const type = b[off];
   378→    if (asciiAt(b, off + 1, 5) !== "CD001") break;
   379→    if (type === 1 && pvd < 0) pvd = off;
   380→    else if (type === 2) joliet = true;
   381→    else if (type === 255) break;
   382→    off += 2048;
   383→  }
   384→  if (pvd < 0) throw new Error("ISO9660: no primary volume descriptor found at sector 16.");
   385→  const p = b.subarray(pvd, pvd + 2048);
   386→  const iso: IsoMeta = {
   387→    sysId: isoStr(b, pvd + 8, 32),
   388→    volId: isoStr(b, pvd + 40, 32),
   389→    volSpace: u32le(p, 80),
   390→    blockSize: u16le(p, 128) || 2048,
   391→    joliet,
   392→  };
   393→  const rootExtent = u32le(p, 158);
   394→  const rootLen = u32le(p, 166);
   395→  const entries: ArcEntry[] = [];
   396→  const warnings: string[] = [];
   397→  let depthCapped = false;
   398→
   399→  const walk = (extent: number, len: number, prefix: string, depth: number) => {
   400→    if (entries.length >= 5000) return;
   401→    const start = extent * iso.blockSize;
   402→    if (start < 0 || start >= b.length) return;
   403→    const end = Math.min(start + len, b.length);
   404→    let o = start;
   405→    while (o + 34 <= end) {
   406→      const recLen = b[o];
   407→      if (recLen === 0) { // records never cross sector boundaries — skip to next sector
   408→        const next = (Math.floor(o / iso.blockSize) + 1) * iso.blockSize;
   409→        if (next <= o) break;
   410→        o = next;
   411→        continue;
   412→      }
   413→      if (recLen < 34 || o + recLen > end) break;
   414→      const ext = u32le(b, o + 2);
   415→      const dlen = u32le(b, o + 10);
   416→      const flags = b[o + 25];
   417→      const nameLen = b[o + 32];
   418→      const name = latin1(b.subarray(o + 33, o + 33 + nameLen));
   419→      o += recLen;
   420→      if (name === "\u0000" || name === "\u0001") continue; // . and ..
   421→      const clean = name.replace(/;\d+$/, "");
   422→      const path = prefix ? prefix + "/" + clean : clean;
   423→      const isDir = (flags & 0x02) !== 0;
   424→      if (isDir) {
   425→        entries.push({
   426→          path, isDir: true, size: 0, packed: 0, method: "directory", crc: null, encrypted: false, mtime: null,
   427→        });
   428→        if (depth < 2) walk(ext, dlen, path, depth + 1);
   429→        else depthCapped = true;
   430→      } else {
   431→        const dataStart = ext * iso.blockSize;
   432→        entries.push({
   433→          path, isDir: false, size: dlen, packed: dlen, method: "ISO9660", crc: null, encrypted: false, mtime: null,
   434→          read: dataStart >= b.length ? undefined
   435→            : () => b.subarray(dataStart, Math.min(dataStart + dlen, b.length)),
   436→        });
   437→      }
   438→    }
   439→  };
   440→  walk(rootExtent, rootLen, "", 0);
   441→  if (depthCapped) warnings.push("Directory walk capped at 2 levels — deeper sub-folders are listed but not expanded.");
   442→  if (joliet) warnings.push("Joliet (UCS-2) supplementary volume detected — showing primary ISO9660 names.");
   443→  return { kind: "iso", entries, iso, warnings };
   444→}

   448→function parseGzHeader(b: Uint8Array): GzMeta {
   449→  if (b.length < 10) throw new Error("GZIP: header truncated.");
   450→  const flg = b[3];
   451→  const mtime = u32le(b, 4);
   452→  let o = 10;
   453→  if (flg & 4) o += 2 + u16le(b, o);
   454→  let originalName: string | undefined;
   455→  if (flg & 8) {
   456→    const e = b.indexOf(0, o);
   457→    if (e > o) originalName = latin1(b.subarray(o, e));
   458→    o = (e < 0 ? o : e) + 1;
   459→  }
   460→  const flags: string[] = [];
   461→  if (flg & 1) flags.push("FTEXT");
   462→  if (flg & 2) flags.push("FHCRC");
   463→  if (flg & 4) flags.push("FEXTRA");
   464→  if (flg & 8) flags.push("FNAME");
   465→  if (flg & 16) flags.push("FCOMMENT");
   466→  return { inner: "", originalName, mtime: mtime ? mtime * 1000 : null, os: GZ_OS[b[9]] ?? `unknown (${b[9]})`, flags };
   467→}

   469→function parseGz(src: ByteSource, fileName: string): ArchiveModel {
   470→  const b = src.bytes;
   471→  const gz = parseGzHeader(b);
   472→  let inner: Uint8Array<ArrayBuffer>;
   473→  try {
   474→    inner = gunzipSync(b);
   475→  } catch (e) {
   476→    throw new Error("GZIP: decompression failed (" + (e instanceof Error ? e.message : String(e)) + "). The stream may be corrupted or multi-member.");
   477→  }
   478→  const warnings: string[] = [];
   479→  if (inner.length >= 262 && asciiAt(inner, 257, 5) === "ustar") {
   480→    const t = parseTar(inner);
   481→    return { kind: "gz", subKind: `TAR (${t.variant})`, entries: t.entries, gz: { ...gz, inner: "TAR archive" }, warnings };
   482→  }
   483→  if (inner[0] === 0x50 && inner[1] === 0x4b) {
   484→    try {
   485→      const z = parseZip({ bytes: inner, base: 0, total: inner.length });
   486→      return { kind: "gz", subKind: "ZIP", entries: z.entries, comment: z.comment, gz: { ...gz, inner: "ZIP archive" }, warnings };
   487→    } catch { /* fall through to raw content */ }
   488→  }
   489→  if (inner.length >= 0x8006 && asciiAt(inner, 0x8001, 5) === "CD001") {
   490→    try {
   491→      const m = parseIso(inner);
   492→      return { kind: "gz", subKind: "ISO", entries: m.entries, gz: { ...gz, inner: "ISO image" }, warnings: m.warnings };
   493→    } catch { /* fall through */ }
   494→  }
   495→  const name = gz.originalName || fileName.replace(/\.(t?gz|gzip)$/i, "") || "content";
   496→  const probe = inner.subarray(0, Math.min(inner.length, 2048));
   497→  const texty = printabilityRatio(probe) > 0.9;
   498→  if (texty) warnings.push("Inner content looks like text — preview uses UTF-8 with CP437 fallback.");
   499→  return {
   500→    kind: "gz",
   501→    subKind: texty ? "text" : "binary",
   502→    entries: [{
   503→      path: name, isDir: false, size: inner.length, packed: b.length, method: "gzip",
   504→      crc: null, encrypted: false, mtime: gz.mtime, read: () => inner,
   505→    }],
   506→    gz: { ...gz, inner: texty ? "Text content" : "Binary content" },
   507→    warnings,
   508→  };
   509→}
   510→
   511→/* =============================== RAR / 7z =============================== */
   512→
   513→function parseRar(b: Uint8Array): ArchiveModel {
   514→  const v5 = b.length > 7 && b[6] === 1;
   515→  return {
   516→    kind: "rar",
   517→    entries: [],
   518→    rar: {
   519→      version: v5 ? "RAR 5.0+" : "RAR 4.x",
   520→      note: v5
   521→        ? "RAR5 uses a proprietary dictionary codec — entry listing and extraction are not available in-browser yet."
   522→        : "RAR4 uses a proprietary compression codec — entry listing and extraction are not available in-browser yet.",
   523→    },
   524→    warnings: [],
   525→  };
   526→}
   527→
   528→function parse7z(b: Uint8Array): ArchiveModel {
   529→  return {
   530→    kind: "7z",
   531→    entries: [],
   532→    z7: {
   533→      version: `${b[6]}.${b[7]}`,
   534→      nextHeaderOffset: u64le(b, 12),
   535→      nextHeaderSize: u64le(b, 20),
   536→    },
   537→    warnings: [],
   538→  };
   539→}
   540→
   541→/* ============================== dispatcher ============================== */
   542→
   543→function parseArchive(src: ByteSource, fileName: string): ArchiveModel {
   544→  const b = src.bytes;
   545→  if (b.length >= 3 && b[0] === 0x50 && b[1] === 0x4b && (b[2] === 3 || b[2] === 5 || b[2] === 7)) {
   546→    return parseZip(src);
   547→  }
   548→  if (b.length >= 7 && asciiAt(b, 0, 4) === "Rar!") return parseRar(b);
   549→  if (b.length >= 6 && asciiAt(b, 0, 2) === "7z" && b[2] === 0xbc && b[3] === 0xaf && b[4] === 0x27 && b[5] === 0x1c) {
   550→    return parse7z(b);
   551→  }
   552→  if (b.length >= 0x8006 && asciiAt(b, 0x8001, 5) === "CD001") return parseIso(b);
   553→  if (b.length >= 2 && b[0] === 0x1f && b[1] === 0x8b) return parseGz(src, fileName);
   554→  if (b.length >= 512 && (asciiAt(b, 257, 5) === "ustar" || tarChecksumOk(b.subarray(0, 512)))) {
   555→    const t = parseTar(b);
   556→    return { kind: "tar", subKind: t.variant, entries: t.entries, warnings: [] };
   557→  }
   558→  throw new Error("Archive structure not recognized (supported listing: ZIP, TAR, GZ/TGZ, ISO 9660; detection only: RAR, 7z).");
   559→}

   660→          <Lock className="mx-auto mb-1 h-4 w-4" />
   661→          Encrypted entry (general-purpose flag bit 0). Password-protected archives cannot be decrypted in this viewer.
   662→        </div>

   805→    setErr(null);
   806→    setSelected(null);

   808→    (async () => {
